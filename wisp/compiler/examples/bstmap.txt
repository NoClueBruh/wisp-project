typedef struct mapnode (
    string key,
    any value,

    mapnode left,
    mapnode right
) mapnode;

const strcmp = L. string str1, string str2: (
    let internal: int -> int = L. int i: (
        alias c1 = str1[i] as int in 
        alias c2 = str2[i] as int in
        
        if i >= str1.length then
            str2.length - i
        else if i >= str2.length then
            i - str1.length
        else (
            if c1 == c2 then 
                internal(i + 1)
            else 
                c1 - c2
        )
    ) in internal(0)
);

const mapnode_value: mapnode -> any = L. mapnode node: node.value;

const map_set: [mapnode, string, any] -> mapnode = L. mapnode node, string key, any val: (
    if ?node then 
        let cmp = strcmp(node.key, key) in 

        if cmp == 0 then
            (node.key, val, node.left, node.right)
        else if cmp > 0 then 
            (node.key, node.value, node.left, map_set(node.right, key, val))
        else 
            (node.key, node.value, map_set(node.left, key, val), node.right)
    else 
        (key, val, null as mapnode, null as mapnode)
);
 
const map_largest: mapnode -> mapnode = L. mapnode node: (
    if ?node && ?node.right then 
        map_largest(node.right)
    else 
        node
);

const map_poplargest: mapnode -> mapnode = L. mapnode node: (
    if ?node then 
        if ?node.right then 
            (node.key, node.value, node.left, map_poplargest(node.right))
        else 
            node.left
    else 
        node
);

const map_remove: [mapnode, string] -> mapnode = L. mapnode node, string key: (
    if ?node then 
        let cmp = strcmp(node.key, key) in 

        if cmp == 0 then 
            if ?node.left then 
                // it was either doing twice the work of popping and getting the largest 
                // or doing it once but allocating more objects (returning multiple values every recursive call bru)
                let largest = map_largest(node.left) in
                (largest.key, largest.value, map_poplargest(node.left), node.right)
            else 
                node.right
        else if cmp > 0 then 
            (node.key, node.value, node.left, map_remove(node.right, key))
        else 
            (node.key, node.value, map_remove(node.left, key), node.right)
    else 
        node
);

const map_new: (list(string, any)) -> mapnode = L. list(string, any) p: (
    let internal: [list(string, any), mapnode] -> mapnode = L. list(string, any) list, mapnode parent:(
        alias pair = $list in

        if ?list then 
            internal(~list, map_set(parent, pair.0, pair.1))
        else 
            parent
    ) in internal(p, null as mapnode)
);

const map_find: [mapnode, string] -> mapnode = L. mapnode node, string key: (
    if ?node then 
        let cmp = strcmp(node.key, key) in 

        if cmp == 0 then  
            node
        else if cmp > 0 then
            map_find(node.right, key)
        else 
            map_find(node.left, key) 
    else 
        node
);

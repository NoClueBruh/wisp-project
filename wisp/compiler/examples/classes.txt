typedef struct integer (
    int num,
    integer -> integer mul,
    integer -> integer add
) integer;

const new_integer: int -> integer = L. int number: (
    number,
    ( L. integer x: new_integer:(number * x.num) ),
    ( L. integer x: new_integer:(number + x.num) )
);

const A = new_integer:30;
const B = new_integer:60;

const C = A.add:B;
const D = C.mul:B;
